Script SQL à exécuter dans le SQL Editor de Supabase : 
CREATE TABLE IF NOT EXISTS randonnees_sanaryens (
  id TEXT PRIMARY KEY,
  name TEXT,
  distance_km NUMERIC,
  elevation_gain NUMERIC,
  elevation_loss NUMERIC,
  min_elevation NUMERIC,
  max_elevation NUMERIC,
  start_lat NUMERIC,
  start_lon NUMERIC,
  estimated_duration TEXT,
  notes TEXT,
  gpx_url TEXT,
  raw_gpx TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);
ALTER TABLE randonnees_sanaryens ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public Read/Write" ON randonnees_sanaryens FOR ALL USING (true) WITH CHECK (true);
